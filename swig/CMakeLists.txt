include(OpenEyeSWIG)

# PYTHON_OUTPUT_DIR below points at python/maptitude, so the module's post-build
# copy lands on the tracked proxy: every build rewrites it with the output of
# whatever SWIG the builder happens to have. A contributor on a different SWIG
# therefore restages a wholly regenerated maptitude.py alongside their real
# change, and nothing in the build says so. Compare the versions at configure
# time, before the build touches the file.
set(_TRACKED_PROXY ${CMAKE_SOURCE_DIR}/python/maptitude/maptitude.py)
if(EXISTS ${_TRACKED_PROXY})
    file(STRINGS ${_TRACKED_PROXY} _PROXY_BANNER REGEX "^# Version " LIMIT_COUNT 1)
    string(REGEX REPLACE "^# Version +" "" _PROXY_SWIG_VERSION "${_PROXY_BANNER}")
    if(_PROXY_SWIG_VERSION AND NOT _PROXY_SWIG_VERSION VERSION_EQUAL SWIG_VERSION)
        message(WARNING
            "SWIG ${SWIG_VERSION} will regenerate python/maptitude/maptitude.py, which "
            "is tracked and was generated by SWIG ${_PROXY_SWIG_VERSION}. Building "
            "overwrites it. Keep that rewrite out of your commit unless regenerating "
            "the proxy is the point, in which case say so in the commit message.")
    endif()
endif()

openeye_add_swig_module(
    NAME maptitude
    SWIG_FILE maptitude.i
    LINK_LIBS maptitude
    PYTHON_OUTPUT_DIR ${CMAKE_SOURCE_DIR}/python/maptitude
    STABLE_ABI ${MAPTITUDE_USE_STABLE_ABI}
    SWIG_FLAGS "-doxygen"
    EXPECTED_LIB_VARS
        OECHEM_LIBRARY OESYSTEM_LIBRARY OEPLATFORM_LIBRARY
        OEMATH_LIBRARY OEGRID_LIBRARY OEFIZZCHEM_LIBRARY
        OEZSTD_LIBRARY
    INIT_PY ${CMAKE_SOURCE_DIR}/python/maptitude/__init__.py
)
